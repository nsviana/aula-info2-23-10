Exercícios de JavaScript

Exemplos da aula com o código-fonte disponível e uma prévia interativa para executar cada exercício.

Os exemplos são carregados das pastas originais. Use os botões nas prévias para conferir o resultado.

1. Lista de cidades do Piauí

Percorre elementos da página: mostra o primeiro item em uma caixa de diálogo e lista todos no console.

HTML — exemplo1/index.html

<html>
<head>
  <meta charset="UTF-8">
</head>
<body>
  <h1>Lista de Cidades do Piaui</h1>
  <ul>
    <li>Campo Maior</li><li>Piripiri</li><li>Picos</li>
    <li>Teresina</li><li>Floriano</li><li>Parnaiba</li>
    <li>Amarante</li><li>Floriano</li><li>Angical</li><li>Coivaras</li>
  </ul>
  <button onclick="mostrarValores()">Clique aqui</button>
  <button onclick="mostrarTodosValores()">Todos</button>
  <script src="script.js"></script>
</body>
</html>

JavaScript — exemplo1/script.js

function mostrarValores(){
  let resultado = document.querySelector("li");
  alert(resultado.innerHTML);
}

function mostrarTodosValores(){
  let resultado = document.querySelectorAll("li");
  for(let i=0; i<resultado.length; i++){
    console.log(resultado[i].innerHTML);
  }
}

Execução

Console

Clique em “Todos” e abra o console do navegador (F12) para ver a lista completa. “Clique aqui” mostra a primeira cidade.

2. Contagem de valores maiores que 10

Percorre um vetor, conta os valores maiores que 10 e apresenta o vetor e a contagem na página.

HTML — questao4/index.html

<!doctype html>
<html lang="pt-BR">
<head><meta charset="UTF-8"></head>
<body>
  <h1>Contagem de valores maiores que 10</h1>
  <p id="valores"></p>
  <p id="resultado" aria-live="polite"></p>
  <button type="button" onclick="mostrarValores()">Ver resultado</button>
  <script src="script.js"></script>
</body>
</html>

JavaScript — questao4/script.js

let numeros = [10, 12, 14, 7, 5, 11, 8, 20, 6];
let contador = 0;

for(let i=0; i<numeros.length; i++){
  if(numeros[i] > 10){
    contador = contador + 1;
  }
}

document.getElementById("valores").textContent = "Valores: " + numeros.join(", ");
document.getElementById("resultado").textContent = "Possui " + contador + " elementos maiores que 10.";
console.log(numeros);
console.log("Possui " + contador + " elementos maiores que 10");

Execução

3. Contagem exibida na página

Versão interativa: ao clicar, apresenta os números do vetor e quantos são maiores que 10.

HTML — questao4_2/index.html

<!doctype html>
<html lang="pt-BR">
<head><meta charset="UTF-8"></head>
<body>
  <p id="valores"></p>
  <p id="resultado"></p>
  <button onclick="mostrarValores()">Clique aqui</button>
  <script src="script.js"></script>
</body>
</html>

JavaScript — questao4_2/script.js

let numeros = [10, 12, 14, 7, 5, 11, 8, 20, 6];

function mostrarValores(){
  let contador = 0;
  let valores = "";

  for(let i=0; i<numeros.length; i++){
    if(numeros[i] > 10){
      contador = contador + 1;
    }
    valores = valores + numeros[i] + "<br>";
  }

  document.getElementById("valores").innerHTML = valores;
  document.getElementById("resultado").textContent = "Possui " + contador + " elementos maiores que 10.";
}
